# 飞书卡片改造实施计划：去标题 + 过程折叠区 + 状态标记 + 双流式

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 飞书回复卡片删除 header，新增默认收起的「思考与工具调用过程」折叠面板与文字状态行，正文与过程区均流式增量更新。

**Architecture:** 单卡三元素（collapsible_panel 内嵌 process markdown + 正文 md + 状态行 status），全部经 cardkit 元素级 API 增量更新；纯函数规划层（cards.ts）与执行层（reply.ts）分离维持现状；outbound 增列过程缓冲。设计 spec：`docs/superpowers/specs/2026-08-25-飞书卡片过程输出与状态标记-design.md`。

**Tech Stack:** TypeScript / vitest / 飞书 cardkit JSON 2.0。

## Global Constraints

- 不自动执行任何 git 提交/变更操作（仓库规则：git mutation 需用户显式确认）。各任务不含 commit 步骤。
- 过程面板标题固定 `思考与工具调用过程`；状态文案固定 `⏳ 输出中…` / `✅ 输出完成` / `❌ 输出出错` / `⏹ 已取消`；工具摘要行格式 `🔧 {name} — {args}`（args 截断 120 字符）。
- 过程区截尾省略标记固定为 `…（已省略前文）\n`。
- 续卡（拆卡后的新卡）结构 = 正文 + 状态行，**不带过程面板**；拆卡后过程区定格在首卡不再更新（`hasPanel` 置 false）。

/** 飞书卡片纯函数：JSON 2.0 构建、UTF-8 字节切分、流式操作规划（拆卡决策全在这里）。 */
import type { TurnStatus } from '../../core/channel.ts'

export const CARD_ELEMENT_ID = 'md'
export const PROCESS_ELEMENT_ID = 'process'
export const STATUS_ELEMENT_ID = 'status'

/** create 未返回真实 card_id 前的占位哨兵（executor 赋值；防并发 flush 重复建卡）。 */
export const PENDING_CARD_ID = '__pending__'

/** 过程区截尾后的头部省略标记。 */
export const PROCESS_OMITTED = '…（已省略前文）\n'

/** 输出中状态行文案。 */
const STATUS_STREAMING = '⏳ 输出中…'

/** 定格状态行文案。 */
const STATUS_FINAL: Record<TurnStatus, string> = {
  done: '✅ 输出完成',
  error: '❌ 输出出错',
  cancelled: '⏹ 已取消',
}

/** 按 UTF-8 字节上限截头（保留头部），不劈开多字节字符与代理对。 */
export function sliceByBytes(text: string, maxBytes: number): string {
  if (Buffer.byteLength(text, 'utf8') <= maxBytes) return text
  let lo = 0
  let hi = text.length
  while (lo < hi) {
    const mid = Math.ceil((lo + hi) / 2)
    if (Buffer.byteLength(text.slice(0, mid), 'utf8') <= maxBytes) lo = mid
    else hi = mid - 1
  }
  let cut = lo
  if (cut > 0) {
    const code = text.charCodeAt(cut - 1)
    if (code >= 0xd8_00 && code <= 0xdb_ff) cut -= 1   // 高位代理在末尾：整对移除
  }
  return text.slice(0, cut)
}

/** 按 UTF-8 字节上限截尾（保留尾部），不劈开多字节字符与代理对；截断时头部加省略标记。 */
export function sliceTailByBytes(text: string, maxBytes: number): string {
  if (Buffer.byteLength(text, 'utf8') <= maxBytes) return text
  const budget = maxBytes - Buffer.byteLength(PROCESS_OMITTED, 'utf8')
  if (budget <= 0) throw new Error(`processMaxBytes=${maxBytes} 过小，连省略标记都容纳不了`)
  let lo = 0
  let hi = text.length
  while (lo < hi) {
    const mid = Math.floor((lo + hi) / 2)
    if (Buffer.byteLength(text.slice(mid), 'utf8') <= budget) hi = mid
    else lo = mid + 1
  }
  let cut = lo
  if (cut < text.length) {
    const code = text.charCodeAt(cut)
    if (code >= 0xdc_00 && code <= 0xdf_ff) cut += 1   // 低位代理在开头：整对移除
  }
  return PROCESS_OMITTED + text.slice(cut)
}

/** 卡片恒以流式模式创建；process 缺省 = 续卡（无过程面板）。 */
export function buildCardJson(opts: { content: string; process?: string }): string {
  return JSON.stringify({
    schema: '2.0',
    config: {
      streaming_mode: true,
      summary: { content: '生成中…' },
      streaming_config: { print_frequency_ms: { default: 70 }, print_step: { default: 1 }, print_strategy: 'fast' },
    },
    body: {
      elements: [
        ...(opts.process !== undefined
          ? [{
              tag: 'collapsible_panel',
              expanded: false,
              header: { title: { tag: 'plain_text', content: '思考与工具调用过程' } },
              elements: [{ tag: 'markdown', content: opts.process, element_id: PROCESS_ELEMENT_ID }],
            }]
          : []),
        { tag: 'markdown', content: opts.content, element_id: CARD_ELEMENT_ID },
        { tag: 'markdown', content: STATUS_STREAMING, element_id: STATUS_ELEMENT_ID },
      ],
    },
  })
}

export interface StreamState {
  /** 当前卡片 id（PENDING_CARD_ID = 创建中）；null = 下一张卡待创建。 */
  cardId: string | null
  /** 当前卡片已消耗的 sequence 计数（create/send 不占）。 */
  seq: number
  /** 当前卡片正文在 answer 中的起始偏移（跨卡累计）。 */
  offset: number
  /** answer 已提交到卡片的前缀长度（跨卡累计）。 */
  shownLen: number
  /** 过程区已提交前缀长度（基于截尾后的 process 文本）。 */
  processShownLen: number
  /** 当前/下一张卡是否带过程面板（仅首卡带；拆卡后 false，过程区定格在首卡）。 */
  hasPanel: boolean
}

export const initialStreamState = (): StreamState => ({
  cardId: null, seq: 0, offset: 0, shownLen: 0, processShownLen: 0, hasPanel: true,
})

export type CardOp =
  | { type: 'create'; cardJson: string }
  | { type: 'send' }
  | { type: 'update'; elementId: string; content: string; sequence: number }
  | { type: 'settings'; streaming: boolean; sequence: number }

/** 把 answer/process 的新增部分同步到卡片；满卡自动关流并开续卡（续卡无面板）。 */
export function planSync(
  state: StreamState,
  answer: string,
  process: string,
  maxBytes: number,
  processMaxBytes: number,
): { state: StreamState; ops: CardOp[] } {
  const proc = sliceTailByBytes(process, processMaxBytes)
  if (answer.length <= state.shownLen && proc.length <= state.processShownLen) return { state, ops: [] }
  const ops: CardOp[] = []
  let { cardId, seq, offset, shownLen, processShownLen, hasPanel } = state
  // 过程区更新只投向带面板的首卡（cardId 指向它时）。
  if (cardId !== null && hasPanel && proc.length > processShownLen) {
    seq += 1
    ops.push({ type: 'update', elementId: PROCESS_ELEMENT_ID, content: proc, sequence: seq })
    processShownLen = proc.length
  }
  while (cardId === null || shownLen < answer.length) {
    const remaining = answer.slice(offset)
    const capacity = sliceByBytes(remaining, maxBytes).length
    if (remaining.length > 0 && capacity <= 0) throw new Error(`cardMaxBytes=${maxBytes} 过小，连一个字符都容纳不了`)
    const fits = remaining.length <= capacity
    const content = fits ? remaining : remaining.slice(0, capacity)
    if (cardId === null) {
      ops.push({ type: 'create', cardJson: buildCardJson({ content, ...(hasPanel ? { process: proc } : {}) }) })
      ops.push({ type: 'send' })
      cardId = PENDING_CARD_ID
      seq = 0
      if (hasPanel) processShownLen = proc.length
    } else if (content.length > shownLen - offset) {
      seq += 1
      ops.push({ type: 'update', elementId: CARD_ELEMENT_ID, content, sequence: seq })
    }
    shownLen = offset + content.length
    if (fits) break
    seq += 1
    ops.push({ type: 'settings', streaming: false, sequence: seq })
    cardId = null
    hasPanel = false
    offset = shownLen
  }
  return { state: { cardId, seq, offset, shownLen, processShownLen, hasPanel }, ops }
}

/** 定格：关流式 + 状态行定格文案。 */
export function planFinalize(state: StreamState, status: TurnStatus): { ops: CardOp[] } {
  if (state.cardId === null) return { ops: [] }
  return {
    ops: [
      { type: 'settings', streaming: false, sequence: state.seq + 1 },
      { type: 'update', elementId: STATUS_ELEMENT_ID, content: STATUS_FINAL[status], sequence: state.seq + 2 },
    ],
  }
}
```

/** 工具调用参数摘要最大字符数。 */
export const TOOL_ARGS_MAX_CHARS = 120

/** 从 assistant 消息内容块提取过程输出：reasoning 全文 + tool_call 摘要行（段落间空行分隔）。 */
export function processOf(content: readonly unknown[]): string {
  const parts: string[] = []
  for (const b of content as readonly { type?: unknown; text?: unknown; name?: unknown; arguments?: unknown }[]) {
    if (b.type === 'reasoning' && typeof b.text === 'string' && b.text.length > 0) {
      parts.push(b.text)
    } else if (b.type === 'tool_call' && typeof b.name === 'string') {
      const args = typeof b.arguments === 'string' ? b.arguments : JSON.stringify(b.arguments ?? {})
      parts.push(`🔧 ${b.name} — ${truncateDetail(args, TOOL_ARGS_MAX_CHARS)}`)
    }
  }
  return parts.length === 0 ? '' : `${parts.join('\n\n')}\n\n`
}
```
